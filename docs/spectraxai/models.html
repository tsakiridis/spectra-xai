spectraxai.models

The module spectraxai.models includes classes and operations on ML models.

This module defines the class StandardModel which should be used to train your models. It supports different ML algorithms and provides short-hand versions to test multiple pre-treatment methods.

 166def _is_default_grid(model: Model, grid: dict[str, Any]) -> bool:
 167    """Whether ``grid`` equals the default grid of ``model`` (e.g. after a clone)."""
 168    default = _DEFAULT_GRIDS.get(model)
 169    if default is None or set(grid) != set(default):
 170        return False
 171    return all(
 172        np.array_equal(np.asarray(grid[k], dtype=object), np.asarray(v, dtype=object))
 173        for k, v in default.items()
 174    )

 177class StandardModel(RegressorMixin, BaseEstimator):
 178    """Class with standard models for ML to apply on spectral datasets.
 179
 180    `StandardModel` is a scikit-learn compatible regressor: besides
 181    ``fit(dataset)`` it accepts ``fit(X, y)``, and it can be used with
 182    `sklearn.base.clone`, pipelines, `sklearn.model_selection.cross_val_score`,
 183    `sklearn.inspection.permutation_importance`, SAGE, etc. For SVR the input
 184    scaling is part of the model, so these tools see the right inputs.
 185
 186    """
 187
 188    model: Model
 189    """The type of `Model` used."""
 190
 191    init_hyperparameters: dict[str, Any]
 192    """A dictionary of the hyperparameters of the models identified by an expert, to
 193    override a grid search."""
 194
 195    grid_search_hyperparameters: dict[str, Any]
 196    """A dictionary containing as keys the hyperparameters of the model and as values a
 197    list of the potential candidate values."""
 198
 199    estimator: Any
 200    """The scikit-learn regressor to use with `Model.CUSTOM` (never modified; a
 201    clone is fitted)."""
 202
 203    cv: int | Iterable[tuple[np.ndarray, np.ndarray]]
 204    """The cross-validation strategy of the grid search."""
 205
 206    n_jobs: int | None
 207    """Number of parallel jobs of the grid search (-1 uses all processors)."""
 208
 209    best_hyperparameters: dict[str, Any] | None
 210    """A dictionary of the best hyperparameters, either set externally or as identified
 211    after calling the train function."""
 212
 213    training_time: float | None
 214    """Training time in seconds."""
 215
 216    testing_time: float | None
 217    """Time for the prediction in seconds."""
 218
 219    best_model: Any
 220    """The best optimized model after tuning the hyperparameters (the bare estimator,
 221    without the SVR input scaling; see `pipeline_`)."""
 222
 223    best_score: float | None
 224    """The best score (R2) in the internal validation set corresponding to the best
 225    model. Only available after a grid search."""
 226
 227    scaler: StandardScaler | None
 228    """The scaler fitted on the training inputs (SVR, the linear models, KNN, GPR
 229    and MLP standardize their inputs)."""
 230
 231    random_state: int | None
 232    """Seed passed to stochastic models (RF, XGBoost, Cubist) and to the SVR gamma
 233    estimation. None keeps the behaviour of earlier versions."""
 234
 235    search: str
 236    """The hyperparameter search strategy: ``"grid"``, ``"random"`` or
 237    ``"bayes"``."""
 238
 239    n_iter: int
 240    """Number of candidates of random and Bayesian search."""
 241
 242    search_space: dict[str, Any] | None
 243    """The space of random and Bayesian search (see `spectraxai.search`), or None
 244    for the model's default space."""
 245
 246    sampler: Any
 247    """The Optuna sampler of Bayesian search, or None for TPE."""
 248
 249    search_space_: dict[str, Any] | None
 250    """The grid or space actually searched by the last call to `fit`, including
 251    any dataset-dependent entries (SVR gamma, PLS n_components)."""
 252
 253    param_grid_: dict[str, Any] | None
 254    """Alias of `search_space_`, kept for compatibility."""
 255
 256    search_results_: pandas.DataFrame | None
 257    """One row per candidate of the last search: its hyperparameters

 266    def __init__(
 267        self,
 268        model: Model,
 269        init_hyperparameters: dict[str, Any] | None = None,
 270        grid_search_hyperparameters: dict[str, Any] | None = None,
 271        *,
 272        estimator: Any = None,
 273        cv: int | Iterable[tuple[np.ndarray, np.ndarray]] = 5,
 274        random_state: int | None = None,
 275        n_jobs: int | None = -1,
 276        search: str = "grid",
 277        n_iter: int = 50,
 278        search_space: dict[str, Any] | None = None,
 279        sampler: Any = None,
 280    ) -> None:
 281        """Initialize StandardModel class for a `Model` and its hyperparameters.
 282
 283        You need to pass either a set of hyperparameters for the model, or a range
 284        thereof in which to search for the optimal set.
 285
 286        Parameters
 287        ----------
 288        model: `Model`
 289            Select a model from `Model` class. Use `Model.CUSTOM` together with
 290            ``estimator`` to bring your own regressor.
 291
 292        init_hyperparameters: `dict`, optional
 293            A dictionary of pre-selected hyperparameters (e.g. a best model)
 294
 295        grid_search_hyperparameters: `dict`, optional
 296            Specify custom grid search range for the hyperparameters
 297
 298        estimator: scikit-learn regressor, optional
 299            The regressor to use with `Model.CUSTOM`, e.g.
 300            ``sklearn.linear_model.Ridge()``. It is cloned, never modified.
 301
 302        cv: int, or an iterable, default=5
 303            The cross-validation strategy of the grid search (see `fit`).
 304
 305        random_state: `int`, optional
 306            Seed for stochastic models (RF, XGBoost, Cubist) and for the SVR gamma
 307            estimation, making results reproducible. Defaults to None, which keeps
 308            the behaviour of earlier versions (XGBoost then uses seed 2000).
 309
 310        n_jobs: `int`, optional
 311            Number of parallel jobs of the grid search. Defaults to -1 (all
 312            processors).
 313
 314        search: `str`, optional
 315            How hyperparameters are tuned when no ``init_hyperparameters`` are
 316            given:
 317
 318            - ``"grid"`` (default): every combination of
 319              ``grid_search_hyperparameters`` (`sklearn.model_selection.GridSearchCV`);
 320            - ``"random"``: ``n_iter`` random candidates from ``search_space``
 321              (`sklearn.model_selection.RandomizedSearchCV`);
 322            - ``"bayes"``: ``n_iter`` Optuna trials over ``search_space``
 323              (`spectraxai.search.BayesianSearchCV`; needs
 324              ``pip install spectraxai[optuna]``).
 325
 326        n_iter: `int`, optional
 327            Number of candidates of random and Bayesian search. Defaults to 50.
 328
 329        search_space: `dict`, optional
 330            The space of random and Bayesian search: ranges such as
 331            ``("log", 0.01, 100)`` or ``("int", 1, 20)``, lists of choices, or
 332            Optuna/SciPy distributions (see `spectraxai.search`). Defaults to
 333            the model's default space, or, if you passed
 334            ``grid_search_hyperparameters``, to those (each list read as choices).
 335
 336        sampler: `optuna.samplers.BaseSampler`, optional
 337            The Optuna sampler of Bayesian search (e.g.
 338            ``optuna.samplers.GPSampler()``). Defaults to TPE seeded by
 339            ``random_state``.
 340
 341        """
 342        if (
 343            init_hyperparameters
 344            and grid_search_hyperparameters
 345            and not _is_default_grid(model, grid_search_hyperparameters)
 346        ):
 347            raise SpectraXAIValueError(
 348                "You cannot specify both init_hyperparameters and "
 349                "grid_search_hyperparameters"
 350            )
 351        if (model == Model.CUSTOM) != (estimator is not None):
 352            raise SpectraXAIValueError(
 353                "Pass an estimator if and only if model is Model.CUSTOM"
 354            )
 355        self.model = model
 356        # Parameters are stored as passed (when given) so that sklearn's clone and
 357        # get_params/set_params work.
 358        self.init_hyperparameters = (
 359            {} if init_hyperparameters is None else init_hyperparameters
 360        )
 361        if grid_search_hyperparameters or model not in _DEFAULT_GRIDS:
 362            self.grid_search_hyperparameters = (
 363                {}
 364                if grid_search_hyperparameters is None
 365                else grid_search_hyperparameters
 366            )
 367        else:
 368            self.grid_search_hyperparameters = dict(_DEFAULT_GRIDS[model])
 369        self.estimator = estimator
 370        self.cv = cv
 371        self.random_state = random_state
 372        self.n_jobs = n_jobs
 373        self.search = search
 374        self.n_iter = n_iter
 375        self.search_space = search_space
 376        self.sampler = sampler
 377        self.training_time = None
 378        self.testing_time = None
 379        self.best_model = None
 380        self.best_hyperparameters = None
 381        self.best_score = None
 382        self.scaler = None
 383        self.search_space_ = None
 384        self.param_grid_ = None
 385        self.search_results_ = None
 386        self.pipeline_ = None

 397    def _new_estimator(self, grid_search: bool) -> Any:
 398        """Create an unfitted estimator for `model`, seeded by `random_state`.
 399
 400        Without a `random_state`, XGBoost keeps the historical seed of 2000 in grid
 401        search and its library default otherwise.
 402
 403        """
 404        seeded: dict[str, Any] = (
 405            {} if self.random_state is None else {"random_state": self.random_state}
 406        )
 407        if self.model == Model.SVR:
 408            return SVR(kernel="rbf", max_iter=int(5e8))
 409        elif self.model == Model.PLS:
 410            return PLSRegression()
 411        elif self.model == Model.RF:
 412            return RandomForestRegressor(**seeded)
 413        elif self.model == Model.CUBIST:
 414            return Cubist(**seeded)
 415        elif self.model == Model.XGBOOST:
 416            if self.random_state is not None:
 417                return xgb.XGBRegressor(random_state=self.random_state)
 418            return xgb.XGBRegressor(seed=2000) if grid_search else xgb.XGBRegressor()
 419        elif self.model == Model.RIDGE:
 420            return Ridge()
 421        elif self.model == Model.LASSO:
 422            return Lasso(max_iter=10000, **seeded)
 423        elif self.model == Model.ELASTIC_NET:
 424            return ElasticNet(max_iter=10000, **seeded)
 425        elif self.model == Model.KNN:
 426            return KNeighborsRegressor()
 427        elif self.model == Model.GPR:
 428            kernel = ConstantKernel() * RBF(length_scale=10.0) + WhiteKernel()
 429            return GaussianProcessRegressor(kernel=kernel, normalize_y=True, **seeded)
 430        elif self.model == Model.MLP:
 431            return MLPRegressor(max_iter=2000, **seeded)
 432        elif self.model == Model.LOCAL_PLS:
 433            return LocalPLSRegressor()
 434        elif self.model == Model.LOCAL_GPR:
 435            return LocalGPRRegressor()
 436        elif self.model == Model.CUSTOM:
 437            return clone(self.estimator)
 438        raise SpectraXAIValueError(f"Unsupported model: {self.model!r}")

 470    def feature_importance(self) -> tuple[np.ndarray | None, str | None]:
 471        """Return the fitted model's own feature importance and how it was computed.
 472
 473        Returns
 474        -------
 475        `tuple`
 476            ``(importance, method)``: VIP scores for PLS (``"VIP"``; for local
 477            PLS the mean VIP of the local models of the training spectra),
 478            ``feature_importances_`` for RF (``"impurity"``), Cubist (``"usage"``),
 479            XGBoost (its ``importance_type``, ``"gain"`` by default) and custom
 480            estimators (``"native"``), absolute coefficients for Ridge, Lasso,
 481            Elastic Net (on standardized inputs) and linear custom estimators
 482            (``"coef"``), or ``(None, None)`` when the model has none
 483            (e.g. SVR; see ``return_permutation_importance`` in
 484            `fit_and_predict`).
 485
 486        """
 487        if self.best_model is None:
 488            raise RuntimeError("You need to train the model first")
 489        model = self.best_model
 490        if self.model == Model.PLS:
 491            return pls_vip(model), "VIP"
 492        if self.model == Model.LOCAL_PLS:
 493            if model.dissimilarity is Dissimilarity.PRECOMPUTED:
 494                return None, None  # needs the matrix: see LocalPLSRegressor.vip
 495            return model.vip(), "VIP"
 496        if self.model == Model.SVR:
 497            return None, None
 498        if hasattr(model, "feature_importances_"):
 499            method = {
 500                Model.RF: "impurity",
 501                Model.CUBIST: "usage",
 502                Model.XGBOOST: getattr(model, "importance_type", None) or "gain",
 503            }.get(self.model, "native")
 504            return np.asarray(model.feature_importances_), method
 505        coef = getattr(model, "coef_", None)
 506        if coef is not None:
 507            coef = np.abs(np.asarray(coef))
 508            if coef.ndim == 2:
 509                coef = coef.mean(axis=0 if coef.shape[1] == self.n_features_in_ else 1)
 510            if coef.size == self.n_features_in_:
 511                return coef.ravel(), "coef"
 512        return None, None

 522    def fit(
 523        self,
 524        dataset: Dataset | Any,
 525        y: Any = None,
 526        cv: int | Iterable[tuple[np.ndarray, np.ndarray]] | None = None,
 527    ) -> "StandardModel":
 528        """Trains the model on a given dataset (or on ``X, y``).
 529
 530        If you didn't supply the `init_hyperparameters` option to the constructor, the
 531        hyperparameters are tuned first, using the cross-validation splitting
 532        strategy specified by cv: a grid search (`sklearn.model_selection.GridSearchCV`)
 533        by default, or a random or Bayesian search (see ``search``). After the
 534        optimal hyperparameters are defined, the model is then retrained on the whole
 535        dataset.
 536
 537        Parameters
 538        ----------
 539        dataset: `spectraxai.dataset.Dataset` or array-like
 540            the Dataset to train the model, or the input matrix ``X`` (then pass
 541            ``y`` too), as scikit-learn does
 542
 543        y: array-like, optional
 544            The output(s), when ``dataset`` is an input matrix
 545
 546        cv: int, or an iterable, optional
 547            Determines the cross-validation splitting strategy. Defaults to the
 548            ``cv`` passed to the constructor (5).
 549            Possible inputs for cv are:
 550            * integer, to specify the number of folds
 551            * An iterable yielding (train, test) splits as arrays of indices
 552
 553        Returns
 554        -------
 555        `StandardModel`
 556            The fitted model (``self``).
 557
 558        """
 559        if isinstance(dataset, Dataset) and y is not None:
 560            # Earlier versions had the signature fit(dataset, cv)
 561            if cv is not None:
 562                raise SpectraXAIValueError("y should not be given with a Dataset")
 563            warnings.warn(
 564                "Passing cv positionally to StandardModel.fit is deprecated; "
 565                "use fit(dataset, cv=...)",
 566                DeprecationWarning,
 567                stacklevel=2,
 568            )
 569            cv, y = y, None
 570        dataset = self._as_dataset(dataset, y)
 571        cv = self.cv if cv is None else cv
 572        if self.search not in _SEARCHES:
 573            raise SpectraXAIValueError(
 574                f"search should be one of {_SEARCHES}, got {self.search!r}"
 575            )
 576        if self.search == "grid" and self.search_space is not None:
 577            raise SpectraXAIValueError(
 578                'search_space is used by search="random" or "bayes"; for a grid '
 579                "search pass grid_search_hyperparameters"
 580            )
 581        if self.model in _SINGLE_OUTPUT_MODELS and dataset.n_outputs > 1:
 582            raise SpectraXAIValueError(
 583                "Cannot create a multi-output model. "
 584                "Please train a model for each output property."
 585            )
 586        self.n_features_in_ = dataset.n_features
 587
 588        # Standardize the inputs of scale-sensitive models
 589        if self.model in _SCALED_MODELS:
 590            self.scaler = StandardScaler()
 591            self.scaler.fit(dataset.X)
 592            X_train = np.asarray(self.scaler.transform(dataset.X))
 593        else:
 594            self.scaler = None
 595            X_train = dataset.X
 596
 597        y_train = np.squeeze(dataset.Y, axis=1) if dataset.n_outputs == 1 else dataset.Y
 598        self.search_results_ = None
 599        # If hyperparameters are passed, use them; otherwise search for them
 600        if len(self.init_hyperparameters) != 0:
 601            self.best_model = self._new_estimator(grid_search=False).set_params(
 602                **self.init_hyperparameters
 603            )
 604            self.search_space_ = self.param_grid_ = None
 605            self.best_score = None
 606            trn_t0 = time.time()
 607            self.best_model.fit(X_train, y_train)
 608            self.best_hyperparameters = dict(self.init_hyperparameters)
 609            trn_t1 = time.time()
 610        else:
 611            space = (
 612                self._grid(X_train)
 613                if self.search == "grid"
 614                else self._search_space(X_train)
 615            )
 616            if any(
 617                Dissimilarity.PRECOMPUTED is v
 618                or (isinstance(v, (list, tuple)) and Dissimilarity.PRECOMPUTED in v)
 619                for v in space.values()
 620            ):
 621                raise SpectraXAIValueError(
 622                    "Hyperparameters cannot be searched with "
 623                    "Dissimilarity.PRECOMPUTED (cross-validation cannot split an "
 624                    "external matrix); pass init_hyperparameters"
 625                )
 626            self.search_space_ = self.param_grid_ = space
 627            estimator = self._new_estimator(grid_search=True)
 628            trn_t0 = time.time()
 629            if self.search == "bayes" and space:
 630                search: Any = BayesianSearchCV(
 631                    estimator,
 632                    space,
 633                    n_iter=self.n_iter,
 634                    cv=cv,
 635                    sampler=self.sampler,
 636                    random_state=self.random_state,
 637                )
 638            elif self.search == "random" and space:
 639                search = RandomizedSearchCV(
 640                    estimator,
 641                    random_distributions(space),
 642                    n_iter=self.n_iter,
 643                    cv=cv,
 644                    n_jobs=self.n_jobs,
 645                    random_state=self.random_state,
 646                )
 647            else:
 648                search = GridSearchCV(
 649                    estimator=estimator,
 650                    param_grid=space,
 651                    cv=cv,
 652                    n_jobs=self.n_jobs,
 653                    verbose=0,
 654                )
 655            search.fit(X_train, y_train)
 656            trn_t1 = time.time()
 657            self.best_model = search.best_estimator_
 658            self.best_hyperparameters = search.best_params_
 659            self.best_score = search.best_score_
 660            self.search_results_ = self._results_table(search)
 661        self.training_time = trn_t1 - trn_t0
 662        steps: list[tuple[str, Any]] = [("model", self.best_model)]
 663        if self.scaler is not None:
 664            steps.insert(0, ("scaler", self.scaler))
 665        self.pipeline_ = Pipeline(steps)
 666        return self

 668    def _grid(self, X_train: np.ndarray) -> dict[str, Any]:
 669        """The grid of a grid search, completed with dataset-dependent entries."""
 670        # Work on a copy so that dataset-dependent entries computed here do not
 671        # leak into later calls (e.g. other folds or pre-processings).
 672        param_grid = dict(self.grid_search_hyperparameters)
 673        if self.model == Model.SVR and "gamma" not in param_grid:
 674            gamma = sigest(X_train, random_state=self.random_state)
 675            param_grid["gamma"] = np.linspace(gamma[0], gamma[2], num=5)
 676        elif self.model == Model.PLS and "n_components" not in param_grid:
 677            param_grid["n_components"] = np.arange(1, min(100, X_train.shape[1]), 1)
 678        return param_grid

 680    def _search_space(self, X_train: np.ndarray) -> dict[str, Any]:
 681        """The space of a random or Bayesian search, with dataset-dependent ranges.
 682
 683        Uses ``search_space`` if given, else custom
 684        ``grid_search_hyperparameters`` (each list read as choices), else the
 685        model's default space.
 686
 687        """
 688        if self.search_space is not None:
 689            space = dict(self.search_space)
 690        elif self.grid_search_hyperparameters and not _is_default_grid(
 691            self.model, self.grid_search_hyperparameters
 692        ):
 693            space = dict(self.grid_search_hyperparameters)
 694        else:
 695            space = dict(_DEFAULT_SPACES.get(self.model, {}))
 696        if self.model == Model.SVR and "gamma" not in space:
 697            gamma = sigest(X_train, random_state=self.random_state)
 698            space["gamma"] = ("log", float(gamma[0]), float(gamma[2]))
 699        elif self.model == Model.PLS and "n_components" not in space:
 700            space["n_components"] = ("int", 1, max(1, min(100, X_train.shape[1]) - 1))
 701        return space

 713    def predict(
 714        self, X_test: np.ndarray | Dataset | Any, dissimilarity: Any = None
 715    ) -> np.ndarray:
 716        """Predict using the best_model from a new unknown input.
 717
 718        Parameters
 719        ----------
 720        X_test: `np.ndarray` or `spectraxai.dataset.Dataset`
 721            The new input data to predict their output
 722
 723        dissimilarity: array-like of shape (n_test_samples, n_training_samples)
 724            For local models with ``dissimilarity=Dissimilarity.PRECOMPUTED``
 725            only: the dissimilarity of each test sample to each training sample
 726            (see `spectraxai.local`).
 727
 728        Returns
 729        -------
 730        `np.ndarray`
 731            A np.ndarray of size (n_test_samples, n_outputs) with the predictions
 732
 733        """
 734        if self.pipeline_ is None:
 735            raise RuntimeError("You need to train the model first")
 736        X = X_test.X if isinstance(X_test, Dataset) else np.asarray(X_test)
 737        if X.ndim == 1:
 738            X = X.reshape(1, -1)
 739        if X.shape[1] != self.n_features_in_:
 740            raise SpectraXAIValueError(
 741                f"X has {X.shape[1]} features, but the model was fitted with "
 742                f"{self.n_features_in_}"
 743            )
 744        tst_t0 = time.time()
 745        params = {} if dissimilarity is None else {"dissimilarity": dissimilarity}
 746        y_hat = self.pipeline_.predict(X, **params)
 747        tst_t1 = time.time()
 748        self.testing_time = tst_t1 - tst_t0
 749        return y_hat

 751    def _permutation_importance(
 752        self, X: np.ndarray, Y: np.ndarray, output: int, dissimilarity: Any = None
 753    ) -> np.ndarray:
 754        """Mean decrease in R2 of output ``output`` when permuting each feature."""
 755        if self.pipeline_ is None:
 756            raise RuntimeError("You need to train the model first")
 757
 758        def score(estimator: Any, X_: np.ndarray, y_: np.ndarray) -> float:
 759            params = {} if dissimilarity is None else {"dissimilarity": dissimilarity}
 760            y_hat = estimator.predict(X_, **params)
 761            y_hat = y_hat[:, output] if y_hat.ndim == 2 else y_hat
 762            return float(r2_score(y_, y_hat))
 763
 764        result = cast(
 765            Bunch,
 766            permutation_importance(
 767                self.pipeline_,
 768                X,
 769                Y[:, output],
 770                scoring=score,
 771                n_repeats=5,
 772                random_state=self.random_state,
 773                n_jobs=self.n_jobs,
 774            ),
 775        )
 776        return np.asarray(result.importances_mean)

 778    def _conformal_intervals(
 779        self, X_trn: np.ndarray, Y_trn: np.ndarray, X_tst: np.ndarray, alpha: float
 780    ) -> tuple[np.ndarray, np.ndarray]:
 781        """CV+ intervals for ``X_tst``, keeping the tuned hyperparameters fixed."""
 782        fixed = cast(StandardModel, clone(self)).set_params(
 783            init_hyperparameters=dict(self.best_hyperparameters or {})
 784        )
 785        cv = self.cv if isinstance(self.cv, int) else 5
 786        conformal = ConformalRegressor(
 787            fixed,
 788            method=ConformalMethod.CV_PLUS,
 789            alpha=alpha,
 790            cv=cv,
 791            random_state=self.random_state,
 792        ).fit(X_trn, Y_trn.ravel() if Y_trn.shape[1] == 1 else Y_trn)
 793        _, lower, upper = conformal.predict(X_tst, return_interval=True)
 794        return lower.reshape(len(X_tst), -1), upper.reshape(len(X_tst), -1)

 796    def fit_and_predict(
 797        self,
 798        dataset: Dataset,
 799        preprocess: SpectralPreprocessingSequence = SpectralPreprocessing.NONE,
 800        idx_trn: np.ndarray | None = None,
 801        idx_tst: np.ndarray | None = None,
 802        get_model: bool = False,
 803        *,
 804        return_permutation_importance: bool = False,
 805        dissimilarity: Any = None,
 806        return_intervals: bool = False,
 807        interval_alpha: float = 0.1,
 808    ) -> pandas.DataFrame:
 809        """Train and test a model in given dataset and spectral pre-processing sequence.
 810
 811        Pass here the whole dataset of (X, Y) and either specify
 812        idx_trn (training indices) or idx_tst (testing indices).
 813
 814        The pre-processing is fitted on the training samples only (this matters for
 815        stateful steps such as `SpectralPreprocessing.MSC`).
 816
 817        Parameters
 818        ----------
 819        dataset: `spectraxai.dataset.Dataset`
 820            the Dataset to train the model
 821
 822        preprocess: `spectraxai.spectra.SpectralPreprocessingSequence`
 823            Optional pre-processing sequence. Defaults to SpectralPreprocessing.NONE.
 824
 825        idx_trn: `np.ndarray`
 826            The indices of the trn samples. Defaults to np.array([]).
 827            This can either be a one dimensional array (for a single split), or
 828            two-dimensional to indicate a cross-validation split.
 829
 830        idx_tst: `np.ndarray`
 831            The indices of the tst samples. Defaults to np.array([]).
 832            This can either be a one dimensional array (for a single split), or
 833            two-dimensional to indicate a cross-validation split.
 834
 835        get_model: `bool`
 836            If true, also return the generated model (column ``model``). Defaults
 837            to False.
 838
 839        return_permutation_importance: `bool`
 840            For models without their own feature importance (e.g. SVR), fill
 841            ``feature_importance`` with the permutation importance on the test set
 842            (mean decrease in R2). Defaults to False.
 843
 844        dissimilarity: array-like of shape (n_samples, n_samples)
 845            For local models with ``dissimilarity=Dissimilarity.PRECOMPUTED``
 846            only: the dissimilarity between every pair of samples of ``dataset``,
 847            e.g. computed from other data than the spectra. The rows of the test
 848            samples and the columns of the training samples are used, and the
 849            feature importance of local PLS is the mean VIP of the local models of
 850            the test samples. Needs fixed hyperparameters (``init_hyperparameters``).
 851
 852        return_intervals: `bool`
 853            Also compute conformal prediction intervals for the test samples
 854            (CV+, see `spectraxai.uncertainty.ConformalRegressor`) with the tuned
 855            hyperparameters, and add the columns ``interval_alpha``,
 856            ``interval_width`` (mean width) and ``interval_coverage`` (fraction of
 857            test values inside their interval). Costs one model fit per fold of
 858            ``cv``. Defaults to False.
 859
 860        interval_alpha: `float`
 861            Miscoverage rate of the intervals: 0.1 (the default) gives 90%
 862            intervals.
 863
 864        Returns
 865        -------
 866        `pandas.DataFrame`
 867            The accuracy results and assorted metadata for each output property.
 868            Every row has ``N``, ``RMSE``, ``R2``, ``RPIQ``, ``output``,
 869            ``pre_process``, ``val_score``, ``model_name``, ``best_hyperparameters``
 870            (dict), ``feature_importance`` (array or None), ``importance_method``,
 871            ``random_state``, ``training_time`` and ``testing_time``, plus the
 872            model's main hyperparameters as separate columns (e.g. ``C``,
 873            ``n_components``) and, for SVR, ``SVs``.
 874
 875        """
 876        idx_trn = np.array([]) if idx_trn is None else idx_trn
 877        idx_tst = np.array([]) if idx_tst is None else idx_tst
 878        if idx_tst.size == 0 and idx_trn.size == 0:
 879            raise SpectraXAIValueError("You need to specify either tst or trn indices")
 880        if idx_tst.size > 0 and idx_trn.size > 0:
 881            raise SpectraXAIValueError("You cannot specify both trn and tst")
 882        if idx_trn.size > 0 and isinstance(idx_trn[0], np.ndarray):  # Multiple-folds
 883            results = []
 884            for fold in range(idx_trn.shape[0]):
 885                result = self.fit_and_predict(
 886                    dataset,
 887                    preprocess,
 888                    idx_trn=idx_trn[fold],
 889                    get_model=get_model,
 890                    return_permutation_importance=return_permutation_importance,
 891                    dissimilarity=dissimilarity,
 892                    return_intervals=return_intervals,
 893                    interval_alpha=interval_alpha,
 894                )
 895                result["fold"] = fold + 1
 896                results.append(result)
 897            return pandas.concat(results, ignore_index=True)
 898        elif idx_tst.size > 0 and isinstance(idx_tst[0], np.ndarray):  # Multiple-folds
 899            results = []
 900            for fold in range(idx_tst.shape[0]):
 901                result = self.fit_and_predict(
 902                    dataset,
 903                    preprocess,
 904                    idx_tst=idx_tst[fold],
 905                    get_model=get_model,
 906                    return_permutation_importance=return_permutation_importance,
 907                    dissimilarity=dissimilarity,
 908                    return_intervals=return_intervals,
 909                    interval_alpha=interval_alpha,
 910                )
 911                result["fold"] = fold + 1
 912                results.append(result)
 913            return pandas.concat(results, ignore_index=True)
 914        elif idx_trn.size > 0 and idx_tst.size == 0:
 915            idx_trn, idx_tst = dataset.train_test_split_explicit(trn=idx_trn)
 916        elif idx_trn.size == 0 and idx_tst.size > 0:
 917            idx_trn, idx_tst = dataset.train_test_split_explicit(tst=idx_tst)
 918
 919        preprocessor = SpectralPreprocessor(preprocess, wavelengths=dataset.wavelengths)
 920        X_trn = preprocessor.fit_transform(dataset.X[idx_trn])
 921        X_tst = preprocessor.transform(dataset.X[idx_tst])
 922        Y_trn, Y_tst = dataset.Y[idx_trn], dataset.Y[idx_tst]
 923        self.fit(Dataset(X_trn, Y_trn))
 924        if self.best_model is None or self.best_hyperparameters is None:
 925            raise RuntimeError("The model failed to fit")
 926        if return_intervals and dissimilarity is not None:
 927            raise SpectraXAIValueError(
 928                "Prediction intervals are not available with a precomputed "
 929                "dissimilarity matrix"
 930            )
 931        D_tst = None
 932        if dissimilarity is not None:
 933            D = np.asarray(dissimilarity)
 934            if D.shape != (dataset.n_samples, dataset.n_samples):
 935                raise SpectraXAIValueError(
 936                    "dissimilarity should be a square matrix over the samples of "
 937                    f"the dataset, of shape {(dataset.n_samples,) * 2}, got {D.shape}"
 938                )
 939            D_tst = D[np.ix_(idx_tst, idx_trn)]
 940        y_hat = self.predict(X_tst, dissimilarity=D_tst)
 941        if D_tst is not None and self.model == Model.LOCAL_PLS:
 942            importance = self.best_model.vip(X_tst, D_tst)
 943            importance_method = "VIP"
 944        else:
 945            importance, importance_method = self.feature_importance()
 946
 947        lower = upper = None
 948        if return_intervals:
 949            lower, upper = self._conformal_intervals(
 950                X_trn, Y_trn, X_tst, interval_alpha
 951            )
 952        results = []
 953        for i in range(dataset.n_outputs):
 954            res = metrics(Y_tst[:, i], y_hat[:, i] if y_hat.ndim == 2 else y_hat)
 955            res["output"] = dataset.Y_names[i]
 956            # Gather all results
 957            res["pre_process"] = str(preprocess)
 958            res["val_score"] = self.best_score
 959            res["model_name"] = str(self.model)
 960            res["best_hyperparameters"] = dict(self.best_hyperparameters)
 961            for key in self._REPORTED_HYPERPARAMETERS[self.model]:
 962                res[key] = self._hyperparameter(key)
 963            if self.model == Model.SVR:
 964                res["SVs"] = len(self.best_model.support_)
 965            if importance is None and return_permutation_importance:
 966                res["feature_importance"] = self._permutation_importance(
 967                    X_tst, Y_tst, i, D_tst
 968                )
 969                res["importance_method"] = "permutation"
 970            else:
 971                res["feature_importance"] = importance
 972                res["importance_method"] = importance_method
 973            if lower is not None and upper is not None:
 974                res["interval_alpha"] = interval_alpha
 975                res["interval_width"] = float(np.mean(upper[:, i] - lower[:, i]))
 976                res["interval_coverage"] = float(
 977                    interval_coverage(Y_tst[:, i], lower[:, i], upper[:, i])[0]
 978                )
 979            res["random_state"] = self.random_state
 980            res["training_time"] = self.training_time
 981            res["testing_time"] = self.testing_time
 982            if get_model:
 983                res["model"] = self.best_model
 984            results.append(res)
 985        return pandas.DataFrame(results)

 398    def _new_estimator(self, grid_search: bool) -> Any:
 399        """Create an unfitted estimator for `model`, seeded by `random_state`.
 400
 401        Without a `random_state`, XGBoost keeps the historical seed of 2000 in grid
 402        search and its library default otherwise.
 403
 404        """
 405        seeded: dict[str, Any] = (
 406            {} if self.random_state is None else {"random_state": self.random_state}
 407        )
 408        if self.model == Model.SVR:
 409            return SVR(kernel="rbf", max_iter=int(5e8))
 410        elif self.model == Model.PLS:
 411            return PLSRegression()
 412        elif self.model == Model.RF:
 413            return RandomForestRegressor(**seeded)
 414        elif self.model == Model.CUBIST:
 415            return Cubist(**seeded)
 416        elif self.model == Model.XGBOOST:
 417            if self.random_state is not None:
 418                return xgb.XGBRegressor(random_state=self.random_state)
 419            return xgb.XGBRegressor(seed=2000) if grid_search else xgb.XGBRegressor()
 420        elif self.model == Model.RIDGE:
 421            return Ridge()
 422        elif self.model == Model.LASSO:
 423            return Lasso(max_iter=10000, **seeded)
 424        elif self.model == Model.ELASTIC_NET:
 425            return ElasticNet(max_iter=10000, **seeded)
 426        elif self.model == Model.KNN:
 427            return KNeighborsRegressor()
 428        elif self.model == Model.GPR:
 429            kernel = ConstantKernel() * RBF(length_scale=10.0) + WhiteKernel()
 430            return GaussianProcessRegressor(kernel=kernel, normalize_y=True, **seeded)
 431        elif self.model == Model.MLP:
 432            return MLPRegressor(max_iter=2000, **seeded)

 434            return LocalPLSRegressor()
 435        elif self.model == Model.LOCAL_GPR:
 436            return LocalGPRRegressor()

 471    def feature_importance(self) -> tuple[np.ndarray | None, str | None]:
 472        """Return the fitted model's own feature importance and how it was computed.
 473
 474        Returns
 475        -------
 476        `tuple`
 477            ``(importance, method)``: VIP scores for PLS (``"VIP"``; for local
 478            PLS the mean VIP of the local models of the training spectra),
 479            ``feature_importances_`` for RF (``"impurity"``), Cubist (``"usage"``),
 480            XGBoost (its ``importance_type``, ``"gain"`` by default) and custom
 481            estimators (``"native"``), absolute coefficients for Ridge, Lasso,
 482            Elastic Net (on standardized inputs) and linear custom estimators
 483            (``"coef"``), or ``(None, None)`` when the model has none
 484            (e.g. SVR; see ``return_permutation_importance`` in
 485            `fit_and_predict`).
 486
 487        """
 488        if self.best_model is None:
 489            raise RuntimeError("You need to train the model first")
 490        model = self.best_model
 491        if self.model == Model.PLS:
 492            return pls_vip(model), "VIP"
 493        if self.model == Model.LOCAL_PLS:
 494            if model.dissimilarity is Dissimilarity.PRECOMPUTED:
 495                return None, None  # needs the matrix: see LocalPLSRegressor.vip
 496            return model.vip(), "VIP"
 497        if self.model == Model.SVR:
 498            return None, None
 499        if hasattr(model, "feature_importances_"):
 500            method = {
 501                Model.RF: "impurity",
 502                Model.CUBIST: "usage",
 503                Model.XGBOOST: getattr(model, "importance_type", None) or "gain",
 504            }.get(self.model, "native")
 505            return np.asarray(model.feature_importances_), method
 506        coef = getattr(model, "coef_", None)
 507        if coef is not None:
 508            coef = np.abs(np.asarray(coef))
 509            if coef.ndim == 2:
 510                coef = coef.mean(axis=0 if coef.shape[1] == self.n_features_in_ else 1)
 511            if coef.size == self.n_features_in_:
 512                return coef.ravel(), "coef"
 513        return None, None

 523    def fit(
 524        self,
 525        dataset: Dataset | Any,
 526        y: Any = None,
 527        cv: int | Iterable[tuple[np.ndarray, np.ndarray]] | None = None,
 528    ) -> "StandardModel":
 529        """Trains the model on a given dataset (or on ``X, y``).
 530
 531        If you didn't supply the `init_hyperparameters` option to the constructor, the
 532        hyperparameters are tuned first, using the cross-validation splitting
 533        strategy specified by cv: a grid search (`sklearn.model_selection.GridSearchCV`)
 534        by default, or a random or Bayesian search (see ``search``). After the
 535        optimal hyperparameters are defined, the model is then retrained on the whole
 536        dataset.
 537
 538        Parameters
 539        ----------
 540        dataset: `spectraxai.dataset.Dataset` or array-like
 541            the Dataset to train the model, or the input matrix ``X`` (then pass
 542            ``y`` too), as scikit-learn does
 543
 544        y: array-like, optional
 545            The output(s), when ``dataset`` is an input matrix
 546
 547        cv: int, or an iterable, optional
 548            Determines the cross-validation splitting strategy. Defaults to the
 549            ``cv`` passed to the constructor (5).
 550            Possible inputs for cv are:
 551            * integer, to specify the number of folds
 552            * An iterable yielding (train, test) splits as arrays of indices
 553
 554        Returns
 555        -------
 556        `StandardModel`
 557            The fitted model (``self``).
 558
 559        """
 560        if isinstance(dataset, Dataset) and y is not None:
 561            # Earlier versions had the signature fit(dataset, cv)
 562            if cv is not None:
 563                raise SpectraXAIValueError("y should not be given with a Dataset")
 564            warnings.warn(
 565                "Passing cv positionally to StandardModel.fit is deprecated; "
 566                "use fit(dataset, cv=...)",
 567                DeprecationWarning,
 568                stacklevel=2,
 569            )
 570            cv, y = y, None
 571        dataset = self._as_dataset(dataset, y)
 572        cv = self.cv if cv is None else cv
 573        if self.search not in _SEARCHES:
 574            raise SpectraXAIValueError(
 575                f"search should be one of {_SEARCHES}, got {self.search!r}"
 576            )
 577        if self.search == "grid" and self.search_space is not None:
 578            raise SpectraXAIValueError(
 579                'search_space is used by search="random" or "bayes"; for a grid '
 580                "search pass grid_search_hyperparameters"
 581            )
 582        if self.model in _SINGLE_OUTPUT_MODELS and dataset.n_outputs > 1:
 583            raise SpectraXAIValueError(
 584                "Cannot create a multi-output model. "
 585                "Please train a model for each output property."
 586            )
 587        self.n_features_in_ = dataset.n_features
 588
 589        # Standardize the inputs of scale-sensitive models
 590        if self.model in _SCALED_MODELS:
 591            self.scaler = StandardScaler()
 592            self.scaler.fit(dataset.X)
 593            X_train = np.asarray(self.scaler.transform(dataset.X))
 594        else:
 595            self.scaler = None
 596            X_train = dataset.X
 597
 598        y_train = np.squeeze(dataset.Y, axis=1) if dataset.n_outputs == 1 else dataset.Y
 599        self.search_results_ = None
 600        # If hyperparameters are passed, use them; otherwise search for them
 601        if len(self.init_hyperparameters) != 0:
 602            self.best_model = self._new_estimator(grid_search=False).set_params(
 603                **self.init_hyperparameters
 604            )
 605            self.search_space_ = self.param_grid_ = None
 606            self.best_score = None
 607            trn_t0 = time.time()
 608            self.best_model.fit(X_train, y_train)
 609            self.best_hyperparameters = dict(self.init_hyperparameters)
 610            trn_t1 = time.time()
 611        else:
 612            space = (
 613                self._grid(X_train)
 614                if self.search == "grid"
 615                else self._search_space(X_train)
 616            )
 617            if any(
 618                Dissimilarity.PRECOMPUTED is v
 619                or (isinstance(v, (list, tuple)) and Dissimilarity.PRECOMPUTED in v)
 620                for v in space.values()
 621            ):
 622                raise SpectraXAIValueError(
 623                    "Hyperparameters cannot be searched with "
 624                    "Dissimilarity.PRECOMPUTED (cross-validation cannot split an "
 625                    "external matrix); pass init_hyperparameters"
 626                )
 627            self.search_space_ = self.param_grid_ = space
 628            estimator = self._new_estimator(grid_search=True)
 629            trn_t0 = time.time()
 630            if self.search == "bayes" and space:
 631                search: Any = BayesianSearchCV(
 632                    estimator,
 633                    space,
 634                    n_iter=self.n_iter,
 635                    cv=cv,
 636                    sampler=self.sampler,
 637                    random_state=self.random_state,
 638                )
 639            elif self.search == "random" and space:
 640                search = RandomizedSearchCV(
 641                    estimator,
 642                    random_distributions(space),
 643                    n_iter=self.n_iter,
 644                    cv=cv,
 645                    n_jobs=self.n_jobs,
 646                    random_state=self.random_state,
 647                )
 648            else:
 649                search = GridSearchCV(
 650                    estimator=estimator,
 651                    param_grid=space,
 652                    cv=cv,
 653                    n_jobs=self.n_jobs,
 654                    verbose=0,
 655                )
 656            search.fit(X_train, y_train)
 657            trn_t1 = time.time()
 658            self.best_model = search.best_estimator_
 659            self.best_hyperparameters = search.best_params_
 660            self.best_score = search.best_score_
 661            self.search_results_ = self._results_table(search)
 662        self.training_time = trn_t1 - trn_t0
 663        steps: list[tuple[str, Any]] = [("model", self.best_model)]
 664        if self.scaler is not None:
 665            steps.insert(0, ("scaler", self.scaler))
 666        self.pipeline_ = Pipeline(steps)
 667        return self

 669    def _grid(self, X_train: np.ndarray) -> dict[str, Any]:
 670        """The grid of a grid search, completed with dataset-dependent entries."""
 671        # Work on a copy so that dataset-dependent entries computed here do not
 672        # leak into later calls (e.g. other folds or pre-processings).
 673        param_grid = dict(self.grid_search_hyperparameters)
 674        if self.model == Model.SVR and "gamma" not in param_grid:
 675            gamma = sigest(X_train, random_state=self.random_state)
 676            param_grid["gamma"] = np.linspace(gamma[0], gamma[2], num=5)
 677        elif self.model == Model.PLS and "n_components" not in param_grid:
 678            param_grid["n_components"] = np.arange(1, min(100, X_train.shape[1]), 1)
 679        return param_grid

 681    def _search_space(self, X_train: np.ndarray) -> dict[str, Any]:
 682        """The space of a random or Bayesian search, with dataset-dependent ranges.
 683
 684        Uses ``search_space`` if given, else custom
 685        ``grid_search_hyperparameters`` (each list read as choices), else the
 686        model's default space.
 687
 688        """
 689        if self.search_space is not None:
 690            space = dict(self.search_space)
 691        elif self.grid_search_hyperparameters and not _is_default_grid(
 692            self.model, self.grid_search_hyperparameters
 693        ):
 694            space = dict(self.grid_search_hyperparameters)
 695        else:
 696            space = dict(_DEFAULT_SPACES.get(self.model, {}))
 697        if self.model == Model.SVR and "gamma" not in space:
 698            gamma = sigest(X_train, random_state=self.random_state)
 699            space["gamma"] = ("log", float(gamma[0]), float(gamma[2]))
 700        elif self.model == Model.PLS and "n_components" not in space:
 701            space["n_components"] = ("int", 1, max(1, min(100, X_train.shape[1]) - 1))
 702        return space

 714    def predict(
 715        self, X_test: np.ndarray | Dataset | Any, dissimilarity: Any = None
 716    ) -> np.ndarray:
 717        """Predict using the best_model from a new unknown input.
 718
 719        Parameters
 720        ----------
 721        X_test: `np.ndarray` or `spectraxai.dataset.Dataset`
 722            The new input data to predict their output
 723
 724        dissimilarity: array-like of shape (n_test_samples, n_training_samples)
 725            For local models with ``dissimilarity=Dissimilarity.PRECOMPUTED``
 726            only: the dissimilarity of each test sample to each training sample
 727            (see `spectraxai.local`).
 728
 729        Returns
 730        -------
 731        `np.ndarray`
 732            A np.ndarray of size (n_test_samples, n_outputs) with the predictions
 733
 734        """
 735        if self.pipeline_ is None:
 736            raise RuntimeError("You need to train the model first")
 737        X = X_test.X if isinstance(X_test, Dataset) else np.asarray(X_test)
 738        if X.ndim == 1:
 739            X = X.reshape(1, -1)
 740        if X.shape[1] != self.n_features_in_:
 741            raise SpectraXAIValueError(
 742                f"X has {X.shape[1]} features, but the model was fitted with "
 743                f"{self.n_features_in_}"
 744            )
 745        tst_t0 = time.time()
 746        params = {} if dissimilarity is None else {"dissimilarity": dissimilarity}
 747        y_hat = self.pipeline_.predict(X, **params)
 748        tst_t1 = time.time()
 749        self.testing_time = tst_t1 - tst_t0
 750        return y_hat
